// apps/web/src/lib/pm-api.ts
const BASE = '/api'

async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    headers: { 'Content-Type': 'application/json' },
    ...init,
  })
  const json = await res.json()
  if (json.error) throw new Error(json.error.message)
  return json.data
}

export const pmApi = {
  projects: {
    list: (params?: Record<string, string>) =>
      apiFetch<Project[]>(`/projects?${new URLSearchParams(params)}`),
    get: (id: string) => apiFetch<ProjectWithProgress>(`/projects/${id}`),
    create: (body: CreateProjectInput) =>
      apiFetch<Project>('/projects', { method: 'POST', body: JSON.stringify(body) }),
    update: (id: string, body: Partial<Project>) =>
      apiFetch<Project>(`/projects/${id}`, { method: 'PATCH', body: JSON.stringify(body) }),
    delete: (id: string) =>
      apiFetch<{ success: boolean }>(`/projects/${id}`, { method: 'DELETE' }),
  },
  tasks: {
    listStatuses: (projectId: string) =>
      apiFetch<TaskStatus[]>(`/projects/${projectId}/tasks/statuses`),
    list: (projectId: string, params?: Record<string, string>) =>
      apiFetch<Task[]>(`/projects/${projectId}/tasks?${new URLSearchParams(params)}`),
    get: (projectId: string, taskId: string) =>
      apiFetch<TaskWithAssignees>(`/projects/${projectId}/tasks/${taskId}`),
    create: (projectId: string, body: CreateTaskInput) =>
      apiFetch<Task>(`/projects/${projectId}/tasks`, { method: 'POST', body: JSON.stringify(body) }),
    update: (projectId: string, taskId: string, body: Partial<Task>) =>
      apiFetch<Task>(`/projects/${projectId}/tasks/${taskId}`, { method: 'PATCH', body: JSON.stringify(body) }),
    delete: (projectId: string, taskId: string) =>
      apiFetch<{ success: boolean }>(`/projects/${projectId}/tasks/${taskId}`, { method: 'DELETE' }),
    listComments: (projectId: string, taskId: string) =>
      apiFetch<Comment[]>(`/projects/${projectId}/tasks/${taskId}/comments`),
    createComment: (projectId: string, taskId: string, body: unknown) =>
      apiFetch<Comment>(`/projects/${projectId}/tasks/${taskId}/comments`, { method: 'POST', body: JSON.stringify({ body }) }),
    myTasks: () => apiFetch<MyTask[]>('/tasks/mine'),
  },
}

'use client'
import { useState } from 'react'
import { useRouter } from 'next/navigation'

const PALETTE = ['#2d6a4f', '#1e3a8a', '#92400e', '#991b1b', '#6b665c', '#7c3aed', '#0369a1', '#b45309']

export default function NewProjectPage() {
  const router = useRouter()
  const [name, setName] = useState('')
  const [color, setColor] = useState(PALETTE[0])
  const [startDate, setStartDate] = useState('')
  const [endDate, setEndDate] = useState('')
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState('')

  async function submit(e: React.FormEvent) {
    e.preventDefault()
    if (!name.trim()) return setError('Name is required')
    setLoading(true)
    try {
      const res = await fetch('/api/projects', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name: name.trim(), color, start_date: startDate || undefined, end_date: endDate || undefined }),
      })
      const json = await res.json()
      if (json.error) throw new Error(json.error.message)
      router.push(`/projects/${json.data.id}/board`)
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to create project')
      setLoading(false)
    }
  }

  return (
    <div className="p-8 max-w-xl mx-auto">
      <h1 style={{ fontFamily: 'Instrument Serif', fontSize: 28, color: 'var(--text)', marginBottom: 8 }}>
        New Project
      </h1>

      <form onSubmit={submit} className="space-y-5 mt-6">
        <div>
          <label style={{ fontFamily: 'DM Sans', fontSize: 13, color: 'var(--text2)', fontWeight: 500 }}>
            Project name *
          </label>
          <input
            value={name}
            onChange={e => setName(e.target.value)}
            placeholder="e.g. Website Redesign"
            className="mt-1.5 w-full px-3 py-2 rounded-lg border text-sm outline-none focus:ring-2 focus:ring-[var(--text)]"

export function KanbanBoard({ projectId, statuses, tasks, onTaskMove, onTaskClick, onAddTask }: KanbanBoardProps) {

'use client'
import { useEffect, useState, useCallback } from 'react'
import { useParams } from 'next/navigation'
import { KanbanBoard } from '@/components/pm/KanbanBoard'
import { TaskDetailPanel } from '@/components/pm/TaskDetailPanel'
import { pmApi } from '@/lib/pm-api'
import type { Task, TaskStatus } from '@/lib/pm-api'

export default function BoardPage() {
  const { id: projectId } = useParams<{ id: string }>()
  const [statuses, setStatuses] = useState<TaskStatus[]>([])
  const [tasks, setTasks] = useState<Task[]>([])
  const [selectedTask, setSelectedTask] = useState<Task | null>(null)
  const [addingToStatus, setAddingToStatus] = useState<string | null>(null)
  const [newTaskTitle, setNewTaskTitle] = useState('')

  const load = useCallback(async () => {
    const [s, t] = await Promise.all([
      pmApi.tasks.listStatuses(projectId),
      pmApi.tasks.list(projectId),
    ])
    setStatuses(s)
    setTasks(t)
  }, [projectId])

  useEffect(() => { load() }, [load])

  async function handleTaskMove(taskId: string, newStatusId: string) {
    setTasks(prev => prev.map(t => t.id === taskId ? { ...t, status_id: newStatusId } : t))
    await pmApi.tasks.update(projectId, taskId, { status_id: newStatusId })
  }

  async function handleAddTask(statusId: string) {
    setAddingToStatus(statusId)
    setNewTaskTitle('')
  }

  async function submitNewTask(e: React.FormEvent) {
    e.preventDefault()
    if (!newTaskTitle.trim() || !addingToStatus) return
    const task = await pmApi.tasks.create(projectId, { title: newTaskTitle.trim(), status_id: addingToStatus })
    setTasks(prev => [...prev, task])
    setAddingToStatus(null)
    setNewTaskTitle('')
  }

  return (
    <div className="flex h-full relative">
      <div className="flex-1 overflow-hidden">
        <KanbanBoard
          projectId={projectId}
          statuses={statuses}
          tasks={tasks}
          onTaskMove={handleTaskMove}
          onTaskClick={setSelectedTask}
          onAddTask={handleAddTask}
        />
      </div>

      {/* Inline add-task input */}
      {addingToStatus && (
        <div className="fixed inset-0 z-10 flex items-center justify-center bg-black/20"
          onClick={() => setAddingToStatus(null)}>
          <form
            onSubmit={submitNewTask}
            onClick={e => e.stopPropagation()}
            className="bg-[var(--surface)] border rounded-xl p-4 shadow-lg w-80"
            style={{ borderColor: 'var(--border)' }}
          >
            <input
              autoFocus
              value={newTaskTitle}
              onChange={e => setNewTaskTitle(e.target.value)}
              placeholder="Task title"
              className="w-full px-3 py-2 rounded-lg border text-sm outline-none"
              style={{ borderColor: 'var(--border)', fontFamily: 'DM Sans', color: 'var(--text)' }}
            />
            <div className="flex gap-2 mt-3">
              <button type="submit" className="px-3 py-1.5 rounded-lg text-sm"
                style={{ background: 'var(--text)', color: '#fff', fontFamily: 'DM Sans' }}>Add</button>
              <button type="button" onClick={() => setAddingToStatus(null)} className="px-3 py-1.5 rounded-lg text-sm"
                style={{ color: 'var(--text2)', fontFamily: 'DM Sans' }}>Cancel</button>
            </div>
          </form>
        </div>
      )}

      {/* Task detail panel */}
      {selectedTask && (
        <TaskDetailPanel
          projectId={projectId}
          task={selectedTask}
          statuses={statuses}
          onClose={() => setSelectedTask(null)}
          onUpdate={async (updates) => {
            const updated = await pmApi.tasks.update(projectId, selectedTask.id, updates)
            setTasks(prev => prev.map(t => t.id === updated.id ? updated : t))
            setSelectedTask(updated)
          }}
        />
      )}
    </div>
  )
}

// apps/web/src/components/pm/TaskDetailPanel.tsx
'use client'
import { useEffect, useState } from 'react'
import { pmApi } from '@/lib/pm-api'
import type { Task, TaskStatus, Comment } from '@/lib/pm-api'

interface TaskDetailPanelProps {
  projectId: string
  task: Task
  statuses: TaskStatus[]
  onClose: () => void
  onUpdate: (updates: Partial<Task>) => Promise<void>
}

const PRIORITIES = ['URGENT', 'HIGH', 'MEDIUM', 'LOW', 'NONE'] as const

export function TaskDetailPanel({ projectId, task, statuses, onClose, onUpdate }: TaskDetailPanelProps) {
  const [comments, setComments] = useState<Comment[]>([])
  const [newComment, setNewComment] = useState('')
  const [title, setTitle] = useState(task.title)
  const [editingTitle, setEditingTitle] = useState(false)

  useEffect(() => {
    pmApi.tasks.listComments(projectId, task.id).then(setComments)
  }, [projectId, task.id])

  async function saveTitle() {
    setEditingTitle(false)
    if (title !== task.title) await onUpdate({ title })
  }

  async function changeStatus(status_id: string) {
    await onUpdate({ status_id })
  }

  async function changePriority(priority: string) {
    await onUpdate({ priority })
  }

  async function submitComment(e: React.FormEvent) {
    e.preventDefault()
    if (!newComment.trim()) return
    const comment = await pmApi.tasks.createComment(projectId, task.id, [{ type: 'text', text: newComment }])
    setComments(prev => [...prev, comment])
    setNewComment('')
  }

  return (
    <div
      className="fixed inset-y-0 right-0 w-[480px] shadow-xl border-l flex flex-col z-20"
      style={{ background: 'var(--surface)', borderColor: 'var(--border)' }}
    >
      {/* Header */}
      <div className="flex items-center justify-between px-5 py-4 border-b" style={{ borderColor: 'var(--border)' }}>
        <span className="text-xs font-medium" style={{ color: 'var(--text3)', fontFamily: 'DM Sans', textTransform: 'uppercase', letterSpacing: '0.05em' }}>
          Task Detail
        </span>
        <button onClick={onClose} style={{ color: 'var(--text2)', fontSize: 18 }}>×</button>
      </div>

      <div className="flex-1 overflow-y-auto p-5 space-y-5">
        {/* Title */}
        {editingTitle ? (
          <input
            autoFocus
            value={title}
            onChange={e => setTitle(e.target.value)}
            onBlur={saveTitle}
            onKeyDown={e => e.key === 'Enter' && saveTitle()}
            className="w-full text-lg font-medium outline-none border-b-2 pb-1"
            style={{ borderColor: 'var(--text)', fontFamily: 'DM Sans', color: 'var(--text)', background: 'transparent' }}
          />
        ) : (
          <h2
            onClick={() => setEditingTitle(true)}
            className="text-lg font-medium cursor-text hover:bg-[var(--surface2)] rounded px-1 -mx-1"
            style={{ fontFamily: 'DM Sans', color: 'var(--text)' }}
          >
            {task.title}
          </h2>
        )}

        {/* Status + Priority */}
        <div className="grid grid-cols-2 gap-3">
          <div>
            <label className="text-xs block mb-1.5" style={{ color: 'var(--text3)', fontFamily: 'DM Sans' }}>Status</label>
            <select
              value={task.status_id}
              onChange={e => changeStatus(e.target.value)}
              className="w-full px-2 py-1.5 rounded-lg border text-sm"
              style={{ borderColor: 'var(--border)', fontFamily: 'DM Sans', color: 'var(--text)', background: 'var(--surface)' }}
            >
              {statuses.map(s => <option key={s.id} value={s.id}>{s.name}</option>)}
            </select>
          </div>
          <div>
            <label className="text-xs block mb-1.5" style={{ color: 'var(--text3)', fontFamily: 'DM Sans' }}>Priority</label>
            <select
              value={task.priority}
              onChange={e => changePriority(e.target.value)}
              className="w-full px-2 py-1.5 rounded-lg border text-sm"
              style={{ borderColor: 'var(--border)', fontFamily: 'DM Sans', color: 'var(--text)', background: 'var(--surface)' }}
            >
              {PRIORITIES.map(p => <option key={p} value={p}>{p}</option>)}
            </select>
          </div>
        </div>

        {/* Due date */}
        <div>
          <label className="text-xs block mb-1.5" style={{ color: 'var(--text3)', fontFamily: 'DM Sans' }}>Due date</label>

            type="datetime-local"
            value={task.due_date ? task.due_date.slice(0, 16) : ''}

            className="px-2 py-1.5 rounded-lg border text-sm"
            style={{ borderColor: 'var(--border)', fontFamily: 'DM Sans', color: 'var(--text)', background: 'var(--surface)' }}
          />
        </div>

        {/* Client visible toggle */}
        <div className="flex items-center gap-3">
          <label className="text-sm" style={{ color: 'var(--text2)', fontFamily: 'DM Sans' }}>Visible in client portal</label>
          <button
            onClick={() => onUpdate({ client_visible: !task.client_visible })}
            className="w-10 h-5 rounded-full transition-colors relative"
            style={{ background: task.client_visible ? 'var(--green)' : 'var(--surface2)' }}
          >
            <div className="w-4 h-4 bg-white rounded-full absolute top-0.5 transition-all shadow"
              style={{ left: task.client_visible ? '22px' : '2px' }} />
          </button>
        </div>

        {/* Comments */}
        <div>
          <h3 className="text-sm font-medium mb-3" style={{ color: 'var(--text)', fontFamily: 'DM Sans' }}>
            Comments ({comments.length})
          </h3>

                </p>
              </div>
            ))}
          </div>
          <form onSubmit={submitComment} className="mt-3 flex gap-2">

              style={{ background: 'var(--text)', color: '#fff', fontFamily: 'DM Sans' }}>Post</button>
          </form>
        </div>
      </div>
    </div>
  )
}

'use client'
import { useEffect, useState, useCallback } from 'react'

const PRIORITY_LABELS: Record<string, { label: string; color: string }> = {
  URGENT: { label: 'Urgent', color: '#991b1b' },
  HIGH: { label: 'High', color: '#92400e' },
  MEDIUM: { label: 'Medium', color: '#1e3a8a' },
  LOW: { label: 'Low', color: '#6b665c' },
  NONE: { label: '—', color: 'var(--text3)' },
}

export default function ListView() {
  const { id: projectId } = useParams<{ id: string }>()
  const [statuses, setStatuses] = useState<TaskStatus[]>([])
  const [tasks, setTasks] = useState<Task[]>([])
  const [selectedTask, setSelectedTask] = useState<Task | null>(null)
  const [filterStatus, setFilterStatus] = useState<string>('all')

  const load = useCallback(async () => {
    const [s, t] = await Promise.all([pmApi.tasks.listStatuses(projectId), pmApi.tasks.list(projectId)])
    setStatuses(s)
    setTasks(t)
  }, [projectId])

  useEffect(() => { load() }, [load])

  const statusMap = Object.fromEntries(statuses.map(s => [s.id, s]))
  const filtered = filterStatus === 'all' ? tasks : tasks.filter(t => t.status_id === filterStatus)

  return (
    <div className="flex h-full">
      <div className="flex-1 p-6 overflow-auto">
        {/* Filter bar */}
        <div className="flex items-center gap-2 mb-4">
          <button
            onClick={() => setFilterStatus('all')}
            className="px-3 py-1.5 rounded-lg text-xs"
            style={{ background: filterStatus === 'all' ? 'var(--text)' : 'var(--surface2)', color: filterStatus === 'all' ? '#fff' : 'var(--text2)', fontFamily: 'DM Sans' }}
          >
            All
          </button>
          {statuses.map(s => (
            <button
              key={s.id}
              onClick={() => setFilterStatus(s.id)}
              className="px-3 py-1.5 rounded-lg text-xs"
              style={{ background: filterStatus === s.id ? s.color : 'var(--surface2)', color: filterStatus === s.id ? '#fff' : 'var(--text2)', fontFamily: 'DM Sans' }}
            >
              {s.name}
            </button>
          ))}
        </div>

        {/* Table */}
        <div className="rounded-xl border overflow-hidden" style={{ borderColor: 'var(--border)' }}>
          <table className="w-full">

              <tr style={{ background: 'var(--surface2)' }}>
                {['Title', 'Status', 'Priority', 'Due', 'Assignees'].map(h => (

                    {h}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              {filtered.map((task, i) => {
                const status = statusMap[task.status_id]
                const priority = PRIORITY_LABELS[task.priority]
                return (
                  <tr
                    key={task.id}
                    onClick={() => setSelectedTask(task)}
                    className="cursor-pointer hover:bg-[var(--surface2)] transition-colors"

      {selectedTask && (

          task={selectedTask}

          onClose={() => setSelectedTask(null)}
          onUpdate={async (updates) => {
            const updated = await pmApi.tasks.update(projectId, selectedTask.id, updates)
            setTasks(prev => prev.map(t => t.id === updated.id ? updated : t))
            setSelectedTask(updated)
          }}
        />
      )}
    </div>
  )
}
