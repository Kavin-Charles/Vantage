import { z } from 'zod'

export const createStatusSchema = z.object({

  title: z.string().min(1).max(500),
  status_id: z.string().uuid(),
  parent_id: z.string().uuid().optional(),
  description: z.unknown().optional(),
  priority: z.enum(['URGENT', 'HIGH', 'MEDIUM', 'LOW', 'NONE']).default('NONE'),
  due_date: z.string().datetime().optional(),
  start_date: z.string().datetime().optional(),
  estimate_hours: z.number().positive().optional(),
  assignee_ids: z.array(z.string().uuid()).optional(),
  label_ids: z.array(z.string().uuid()).optional(),
  client_visible: z.boolean().default(false),
})

export const updateTaskSchema = z.object({
  title: z.string().min(1).max(500).optional(),
  status_id: z.string().uuid().optional(),
  parent_id: z.string().uuid().nullable().optional(),
  description: z.unknown().optional(),
  priority: z.enum(['URGENT', 'HIGH', 'MEDIUM', 'LOW', 'NONE']).optional(),
  due_date: z.string().datetime().nullable().optional(),
  start_date: z.string().datetime().nullable().optional(),
  estimate_hours: z.number().positive().nullable().optional(),
  assignee_ids: z.array(z.string().uuid()).optional(),
  label_ids: z.array(z.string().uuid()).optional(),
  client_visible: z.boolean().optional(),
  position: z.number().optional(),
})

export const bulkUpdateTaskSchema = z.object({
  ids: z.array(z.string().uuid()).min(1).max(100),
  status_id: z.string().uuid().optional(),
  priority: z.enum(['URGENT', 'HIGH', 'MEDIUM', 'LOW', 'NONE']).optional(),
  assignee_ids: z.array(z.string().uuid()).optional(),
})

export const listTasksSchema = z.object({
  status_id: z.string().uuid().optional(),
  priority: z.enum(['URGENT', 'HIGH', 'MEDIUM', 'LOW', 'NONE']).optional(),
  assignee_id: z.string().uuid().optional(),
  parent_id: z.string().uuid().nullable().optional(),
  search: z.string().optional(),
  overdue: z.coerce.boolean().optional(),
})

export const createCommentSchema = z.object({
  body: z.unknown(),
  parent_id: z.string().uuid().optional(),
})

export type CreateStatusInput = z.infer<typeof createStatusSchema>
export type CreateTaskInput = z.infer<typeof createTaskSchema>
export type UpdateTaskInput = z.infer<typeof updateTaskSchema>
export type BulkUpdateTaskInput = z.infer<typeof bulkUpdateTaskSchema>
export type ListTasksInput = z.infer<typeof listTasksSchema>
export type CreateCommentInput = z.infer<typeof createCommentSchema>

// apps/api/src/modules/project-tasks/project-tasks.queries.ts
import { db } from '../../db'
import type { CreateStatusInput, CreateTaskInput, UpdateTaskInput, ListTasksInput, CreateCommentInput } from './project-tasks.schema'

const DEFAULT_STATUSES = [
  { name: 'Backlog', color: '#9e998f', position: 0, is_done: false },
  { name: 'In Progress', color: '#1e3a8a', position: 1, is_done: false },
  { name: 'In Review', color: '#92400e', position: 2, is_done: false },
  { name: 'Done', color: '#2d6a4f', position: 3, is_done: true },
]

export async function seedDefaultStatuses(projectId: string) {
  await db
    .insertInto('project_task_statuses')
    .values(DEFAULT_STATUSES.map(s => ({ ...s, project_id: projectId })))
    .execute()
}

export async function listStatuses(projectId: string) {
  return db
    .selectFrom('project_task_statuses')
    .selectAll()
    .where('project_id', '=', projectId)
    .orderBy('position', 'asc')
    .execute()
}

export async function createStatus(projectId: string, input: CreateStatusInput) {
  return db
    .insertInto('project_task_statuses')
    .values({ project_id: projectId, ...input })
    .returningAll()
    .executeTakeFirstOrThrow()
}

export async function listTasks(projectId: string, filters: ListTasksInput) {
  let query = db
    .selectFrom('project_tasks as t')
    .selectAll('t')
    .where('t.project_id', '=', projectId)

  if (filters.status_id) query = query.where('t.status_id', '=', filters.status_id)
  if (filters.priority) query = query.where('t.priority', '=', filters.priority)
  if (filters.parent_id !== undefined) {
    query = filters.parent_id === null
      ? query.where('t.parent_id', 'is', null)
      : query.where('t.parent_id', '=', filters.parent_id)
  }
  if (filters.overdue) {
    query = query
      .where('t.due_date', '<', new Date().toISOString())
      .innerJoin('project_task_statuses as s', 's.id', 't.status_id')
      .where('s.is_done', '=', false)
  }
  if (filters.search) query = query.where('t.title', 'ilike', `%${filters.search}%`)
  if (filters.assignee_id) {
    query = query
      .innerJoin('project_task_assignees as a', 'a.task_id', 't.id')
      .where('a.user_id', '=', filters.assignee_id)
  }

  return query.orderBy('t.position', 'asc').execute()
}

export async function getTask(id: string, projectId: string) {
  return db
    .selectFrom('project_tasks')
    .selectAll()
    .where('id', '=', id)
    .where('project_id', '=', projectId)
    .executeTakeFirst()
}

export async function createTask(projectId: string, userId: string, input: CreateTaskInput) {
  const task = await db
    .insertInto('project_tasks')
    .values({
      project_id: projectId,
      created_by: userId,
      title: input.title,
      status_id: input.status_id,
      parent_id: input.parent_id ?? null,
      description: input.description ? JSON.stringify(input.description) : null,
      priority: input.priority,
      due_date: input.due_date ?? null,
      start_date: input.start_date ?? null,
      estimate_hours: input.estimate_hours ?? null,
      client_visible: input.client_visible,
    })
    .returningAll()
    .executeTakeFirstOrThrow()

  if (input.assignee_ids?.length) {
    await db.insertInto('project_task_assignees')
      .values(input.assignee_ids.map(uid => ({ task_id: task.id, user_id: uid })))
      .execute()
  }
  if (input.label_ids?.length) {
    await db.insertInto('project_task_label_assignments')
      .values(input.label_ids.map(lid => ({ task_id: task.id, label_id: lid })))
      .execute()
  }
  return task
}

export async function updateTask(id: string, projectId: string, input: UpdateTaskInput) {
  const updates: Record<string, unknown> = { updated_at: new Date().toISOString() }
  if (input.title !== undefined) updates.title = input.title
  if (input.status_id !== undefined) updates.status_id = input.status_id
  if (input.parent_id !== undefined) updates.parent_id = input.parent_id
  if (input.description !== undefined) updates.description = input.description ? JSON.stringify(input.description) : null
  if (input.priority !== undefined) updates.priority = input.priority
  if (input.due_date !== undefined) updates.due_date = input.due_date
  if (input.start_date !== undefined) updates.start_date = input.start_date
  if (input.estimate_hours !== undefined) updates.estimate_hours = input.estimate_hours
  if (input.client_visible !== undefined) updates.client_visible = input.client_visible
  if (input.position !== undefined) updates.position = input.position

  const task = await db
    .updateTable('project_tasks')
    .set(updates)
    .where('id', '=', id)
    .where('project_id', '=', projectId)
    .returningAll()
    .executeTakeFirstOrThrow()

  if (input.assignee_ids !== undefined) {
    await db.deleteFrom('project_task_assignees').where('task_id', '=', id).execute()
    if (input.assignee_ids.length) {
      await db.insertInto('project_task_assignees')
        .values(input.assignee_ids.map(uid => ({ task_id: id, user_id: uid })))
        .execute()
    }
  }
  if (input.label_ids !== undefined) {
    await db.deleteFrom('project_task_label_assignments').where('task_id', '=', id).execute()
    if (input.label_ids.length) {
      await db.insertInto('project_task_label_assignments')
        .values(input.label_ids.map(lid => ({ task_id: id, label_id: lid })))
        .execute()
    }
  }
  return task
}

export async function deleteTask(id: string, projectId: string) {
  await db.deleteFrom('project_tasks').where('id', '=', id).where('project_id', '=', projectId).execute()
}

export async function getTaskAssignees(taskId: string) {
  return db
    .selectFrom('project_task_assignees as a')
    .innerJoin('users as u', 'u.id', 'a.user_id')
    .select(['u.id', 'u.name', 'u.email'])
    .where('a.task_id', '=', taskId)
    .execute()
}

export async function listComments(taskId: string) {
  return db
    .selectFrom('project_task_comments as c')
    .leftJoin('users as u', 'u.id', 'c.user_id')
    .select(['c.id', 'c.task_id', 'c.body', 'c.parent_id', 'c.created_at', 'c.updated_at', 'c.portal_session_id', 'u.name as author_name', 'u.email as author_email'])
    .where('c.task_id', '=', taskId)
    .orderBy('c.created_at', 'asc')
    .execute()
}

export async function createComment(taskId: string, userId: string, input: CreateCommentInput) {
  return db
    .insertInto('project_task_comments')
    .values({
      task_id: taskId,
      user_id: userId,
      body: JSON.stringify(input.body),
      parent_id: input.parent_id ?? null,
    })
    .returningAll()
    .executeTakeFirstOrThrow()
}

export async function getTasksForUser(workspaceId: string, userId: string) {
  return db
    .selectFrom('project_tasks as t')
    .innerJoin('project_task_assignees as a', 'a.task_id', 't.id')
    .innerJoin('projects as p', 'p.id', 't.project_id')
    .innerJoin('project_task_statuses as s', 's.id', 't.status_id')
    .select(['t.id', 't.title', 't.due_date', 't.priority', 't.project_id', 'p.name as project_name', 's.name as status_name', 's.is_done'])
    .where('p.workspace_id', '=', workspaceId)
    .where('a.user_id', '=', userId)
    .where('s.is_done', '=', false)
    .where('p.status', '!=', 'DELETED')
    .orderBy('t.due_date', 'asc')
    .execute()
}

// apps/api/src/modules/project-tasks/project-tasks.controller.ts
import type { Request, Response } from 'express'
import * as q from './project-tasks.queries'
import {
  createTaskSchema, updateTaskSchema, bulkUpdateTaskSchema,
  listTasksSchema, createCommentSchema, createStatusSchema
} from './project-tasks.schema'

export async function listStatuses(req: Request, res: Response) {
  const statuses = await q.listStatuses(req.params.projectId)
  return res.json({ data: statuses, error: null })
}

export async function createStatus(req: Request, res: Response) {
  const parsed = createStatusSchema.safeParse(req.body)
  if (!parsed.success) return res.status(400).json({ data: null, error: { code: 'VALIDATION', message: parsed.error.message } })
  const status = await q.createStatus(req.params.projectId, parsed.data)
  return res.status(201).json({ data: status, error: null })
}

export async function list(req: Request, res: Response) {
  const parsed = listTasksSchema.safeParse(req.query)
  if (!parsed.success) return res.status(400).json({ data: null, error: { code: 'VALIDATION', message: parsed.error.message } })
  const tasks = await q.listTasks(req.params.projectId, parsed.data)
  return res.json({ data: tasks, error: null })
}

export async function get(req: Request, res: Response) {
  const task = await q.getTask(req.params.taskId, req.params.projectId)
  if (!task) return res.status(404).json({ data: null, error: { code: 'NOT_FOUND', message: 'Task not found' } })
  const assignees = await q.getTaskAssignees(task.id)
  return res.json({ data: { ...task, assignees }, error: null })
}

export async function create(req: Request, res: Response) {
  const parsed = createTaskSchema.safeParse(req.body)
  if (!parsed.success) return res.status(400).json({ data: null, error: { code: 'VALIDATION', message: parsed.error.message } })
  const task = await q.createTask(req.params.projectId, req.user.id, parsed.data)
  return res.status(201).json({ data: task, error: null })
}

export async function update(req: Request, res: Response) {
  const parsed = updateTaskSchema.safeParse(req.body)
  if (!parsed.success) return res.status(400).json({ data: null, error: { code: 'VALIDATION', message: parsed.error.message } })
  try {
    const task = await q.updateTask(req.params.taskId, req.params.projectId, parsed.data)
    return res.json({ data: task, error: null })
  } catch {
    return res.status(404).json({ data: null, error: { code: 'NOT_FOUND', message: 'Task not found' } })
  }
}

export async function remove(req: Request, res: Response) {
  await q.deleteTask(req.params.taskId, req.params.projectId)
  return res.json({ data: { success: true }, error: null })
}

export async function bulk(req: Request, res: Response) {
  const parsed = bulkUpdateTaskSchema.safeParse(req.body)
  if (!parsed.success) return res.status(400).json({ data: null, error: { code: 'VALIDATION', message: parsed.error.message } })
  const { ids, ...updates } = parsed.data
  if (updates.status_id) {
    await Promise.all(ids.map(id => q.updateTask(id, req.params.projectId, { status_id: updates.status_id })))
  }
  if (updates.priority) {
    await Promise.all(ids.map(id => q.updateTask(id, req.params.projectId, { priority: updates.priority })))
  }
  return res.json({ data: { updated: ids.length }, error: null })
}

export async function listComments(req: Request, res: Response) {
  const comments = await q.listComments(req.params.taskId)
  return res.json({ data: comments, error: null })
}

export async function createComment(req: Request, res: Response) {
  const parsed = createCommentSchema.safeParse(req.body)
  if (!parsed.success) return res.status(400).json({ data: null, error: { code: 'VALIDATION', message: parsed.error.message } })
  const comment = await q.createComment(req.params.taskId, req.user.id, parsed.data)
  return res.status(201).json({ data: comment, error: null })
}

export async function myTasks(req: Request, res: Response) {
  const tasks = await q.getTasksForUser(req.workspace.id, req.user.id)
  return res.json({ data: tasks, error: null })
}

import { Router } from 'express'
import * as ctrl from './project-tasks.controller'

const router = Router({ mergeParams: true })

// statuses
router.get('/statuses', ctrl.listStatuses)
router.post('/statuses', ctrl.createStatus)

// tasks
router.get('/', ctrl.list)
router.post('/', ctrl.create)
router.post('/bulk', ctrl.bulk)
router.get('/:taskId', ctrl.get)
router.patch('/:taskId', ctrl.update)
router.delete('/:taskId', ctrl.remove)

// comments
router.get('/:taskId/comments', ctrl.listComments)
router.post('/:taskId/comments', ctrl.createComment)

export default router

// apps/api/src/__tests__/project-tasks.test.ts
import { describe, it, expect, beforeEach } from 'vitest'
import request from 'supertest'
import app from '../index'
import { getTestToken, getTestWorkspaceId, createTestProject, cleanupProjects } from './helpers'

describe('Project Tasks API', () => {
  let token: string
  let projectId: string

  beforeEach(async () => {
    token = await getTestToken()
    const workspaceId = await getTestWorkspaceId()
    await cleanupProjects(workspaceId)
    const project = await createTestProject(token)
    projectId = project.id
  })

  it('GET /api/projects/:id/tasks/statuses returns default statuses', async () => {
    const res = await request(app)
      .get(`/api/projects/${projectId}/tasks/statuses`)
      .set('Authorization', `Bearer ${token}`)
    expect(res.status).toBe(200)
    expect(res.body.data.length).toBe(4)
    expect(res.body.data[0].name).toBe('Backlog')
    expect(res.body.data[3].is_done).toBe(true)
  })

  it('POST /api/projects/:id/tasks creates task', async () => {
    const statuses = await request(app)
      .get(`/api/projects/${projectId}/tasks/statuses`)
      .set('Authorization', `Bearer ${token}`)
    const statusId = statuses.body.data[0].id

    const res = await request(app)
      .post(`/api/projects/${projectId}/tasks`)
      .set('Authorization', `Bearer ${token}`)
      .send({ title: 'Build feature', status_id: statusId })
    expect(res.status).toBe(201)
    expect(res.body.data.title).toBe('Build feature')
    expect(res.body.data.priority).toBe('NONE')
  })

  it('PATCH /api/projects/:id/tasks/:taskId updates status', async () => {
    const statuses = await request(app)
      .get(`/api/projects/${projectId}/tasks/statuses`)
      .set('Authorization', `Bearer ${token}`)
    const [backlog, , , done] = statuses.body.data

    const created = await request(app)
      .post(`/api/projects/${projectId}/tasks`)
      .set('Authorization', `Bearer ${token}`)
      .send({ title: 'Test task', status_id: backlog.id })

    const res = await request(app)
      .patch(`/api/projects/${projectId}/tasks/${created.body.data.id}`)
      .set('Authorization', `Bearer ${token}`)
      .send({ status_id: done.id })
    expect(res.status).toBe(200)
    expect(res.body.data.status_id).toBe(done.id)
  })

  it('POST /api/projects/:id/tasks/bulk updates multiple tasks', async () => {
    const statuses = await request(app)
      .get(`/api/projects/${projectId}/tasks/statuses`)
      .set('Authorization', `Bearer ${token}`)
    const statusId = statuses.body.data[0].id
    const doneId = statuses.body.data[3].id

    const t1 = await request(app).post(`/api/projects/${projectId}/tasks`).set('Authorization', `Bearer ${token}`).send({ title: 'T1', status_id: statusId })
    const t2 = await request(app).post(`/api/projects/${projectId}/tasks`).set('Authorization', `Bearer ${token}`).send({ title: 'T2', status_id: statusId })

    const res = await request(app)
      .post(`/api/projects/${projectId}/tasks/bulk`)
      .set('Authorization', `Bearer ${token}`)
      .send({ ids: [t1.body.data.id, t2.body.data.id], status_id: doneId })
    expect(res.status).toBe(200)
    expect(res.body.data.updated).toBe(2)
  })

  it('GET /api/tasks/mine returns cross-project tasks for user', async () => {
    const res = await request(app)
      .get('/api/tasks/mine')
      .set('Authorization', `Bearer ${token}`)
    expect(res.status).toBe(200)
    expect(Array.isArray(res.body.data)).toBe(true)
  })
})
